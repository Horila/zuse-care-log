/* ==========================================================================
 * PASTE THIS OVER THE BOTTOM OF YOUR APPS SCRIPT
 *
 * This is everything from "function doPost(e) {" to the end of the file.
 * It deliberately starts below your SECRET, so your own secret is never
 * touched.
 *
 * HOW:
 * 1. Open the Apps Script editor on the sheet.
 * 2. Click on the line that reads   function doPost(e) {
 *    Select from there all the way to the very bottom of the file, and
 *    delete it. Everything above that line stays exactly as it is.
 * 3. Paste this whole file in its place. Save.
 *
 * 4. THE WEB APP MUST BE REDEPLOYED, or the app's new buttons will answer
 *    "unknown action". Deploy -> Manage deployments -> the pencil icon ->
 *    Version: New version -> Deploy. The URL does not change, so syncing
 *    carries on working.
 *
 * 5. Sync once from the app. That creates the "Stock" tab on the sheet.
 *
 * 6. In the editor, pick "installStockAlerts" from the function dropdown and
 *    Run it once. This is the run that asks for calendar permission - say yes.
 *    It is done by hand, from the editor, on purpose: no web app path ever
 *    touches the calendar, so syncing cannot break if that permission is
 *    missing or later withdrawn.
 *
 * ALERT_EMAIL must be filled in for any of the email or calendar reminders to
 * happen -- the monthly report button included. It sits BELOW the cut line, so
 * pasting this over the bottom of your script wipes it back to blank. Copy your
 * address out first, and put it back into the ALERT_EMAIL line of this block
 * after pasting, before you save and redeploy.
 * ========================================================================== */

function doPost(e) {
  let body;
  try {
    body = JSON.parse(e.postData.contents);
  } catch (err) {
    return jsonOut({ error: 'bad json' });
  }
  if (!body.secret || body.secret !== SECRET) {
    return jsonOut({ error: 'unauthorized' });
  }
  if (body.action === 'append') {
    const added = appendRows(body.rows || []);
    return jsonOut({ added: added });
  }
  // The app's "email the report now" button.
  if (body.action === 'report') {
    if (!ALERT_EMAIL) return jsonOut({ error: 'ALERT_EMAIL is blank in the script' });
    sendMonthlyReport();
    return jsonOut({ ok: true, to: ALERT_EMAIL });
  }
  // Stock levels are kept in the app; this mirrors them onto a sheet tab so the
  // daily checker can see them, and so they are visible to whoever opens the
  // sheet. No CalendarApp call may ever appear on a doPost path: an ungranted
  // calendar scope would then break syncing, which is used every day.
  if (body.action === 'stock') {
    const n = writeStockTab_(body.items || []);
    return jsonOut({ ok: true, saved: n });
  }
  // The "cancel vet email" button: preemptively sets the exact dedupe key
  // checkStock() checks, so it works whether clicked before or after the
  // threshold is actually reached. No CalendarApp call, so it's fine here.
  if (body.action === 'skipVetOrder') {
    if (!body.name || !body.since) return jsonOut({ error: 'missing name/since' });
    PropertiesService.getScriptProperties().setProperty('vetorder:' + body.name + '|' + body.since, '1');
    return jsonOut({ ok: true });
  }
  return jsonOut({ error: 'unknown action' });
}

function getSheet_() {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  if (!sh) throw new Error('Sheet "' + SHEET_NAME + '" not found');
  return sh;
}

/** Reads every data row (skipping the 2 header rows), carrying the date
 *  forward for rows where the DATE cell is blank (same-day continuation). */
function readRows() {
  const sh = getSheet_();
  const values = sh.getDataRange().getValues();
  const out = [];
  let curDate = '';
  for (let i = 2; i < values.length; i++) {
    const row = values[i];
    const dateCell = row[1];
    const timeCell = row[2];
    const type = row[3];
    const qty = row[4];
    const notes = row[5];
    if (dateCell) curDate = formatDate_(dateCell);
    if (!type) continue; // blank/free-text rows with no TYPE are skipped
    out.push({
      date: curDate,
      time: formatTime_(timeCell),
      type: String(type),
      qty: (qty === '' || qty == null) ? '' : qty,
      notes: notes ? String(notes) : ''
    });
  }
  return out;
}

/** Appends rows at the bottom, keeping the "blank date = same day as
 *  previous row" convention used throughout the sheet. */
function appendRows(rows) {
  const sh = getSheet_();
  const lastRow = sh.getLastRow();
  let lastDate = '';
  if (lastRow >= 3) {
    for (let r = lastRow; r >= 3; r--) {
      const d = sh.getRange(r, 2).getValue();
      if (d) { lastDate = formatDate_(d); break; }
    }
  }
  const out = [];
  rows.forEach(function (r) {
    const sameDate = (r.date === lastDate);
    out.push(['', sameDate ? '' : r.date, r.time, r.type, r.qty, r.notes || '']);
    lastDate = r.date;
  });
  if (out.length) {
    sh.getRange(sh.getLastRow() + 1, 1, out.length, 6).setValues(out);
  }
  return out.length;
}

function formatDate_(v) {
  if (Object.prototype.toString.call(v) === '[object Date]') {
    return Utilities.formatDate(v, Session.getScriptTimeZone(), 'dd/MM/yyyy');
  }
  return String(v).trim();
}
function formatTime_(v) {
  if (Object.prototype.toString.call(v) === '[object Date]') {
    return Utilities.formatDate(v, Session.getScriptTimeZone(), 'h:mm a').toLowerCase();
  }
  return String(v).trim().toLowerCase();
}

function jsonOut(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

/* ==========================================================================
 * MISSED-SHOT REMINDER
 *
 * The app itself cannot do this. A PWA has no reliable way to wake itself up
 * in the background — a service worker is stopped when the phone sleeps, and
 * real web push needs a push server. This script is already deployed and
 * already runs on Google's clock, so the reminder lives here instead.
 *
 * SETUP (one-time, after pasting this file in):
 * 1. Set ALERT_EMAIL below to the address you actually read on your phone.
 * 2. Check AM_TIME / PM_TIME match the times set in the app's Setup screen.
 * 3. In the Apps Script editor, pick "installReminders" from the function
 *    dropdown and click Run. Authorize when asked.
 * 4. That's it — it now checks every 15 minutes and emails you once per
 *    missed slot. Run "installReminders" again any time you change the times;
 *    it clears the old trigger first, so it never doubles up.
 *
 * To turn it off: Apps Script editor -> Triggers (clock icon) -> delete.
 * ========================================================================== */

const ALERT_EMAIL = '';            // e.g. 'you@example.com' — blank disables alerts
const AM_TIME = '11:30';           // 24h, must match the app's Setup screen
const PM_TIME = '23:30';
const GRACE_MIN = 30;              // how late a shot can be before you're told
const ALERT_WINDOW_H = 5;          // stop nagging about a slot this long after it
const LOOKBACK_H = 2;              // a shot this far before the slot still counts

/** Installs (or reinstalls) the 15-minute checker. Run this by hand once. */
function installReminders() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'checkShotDue') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('checkShotDue').timeBased().everyMinutes(15).create();
  const who = ALERT_EMAIL || '(nobody — ALERT_EMAIL is blank)';
  Logger.log('Reminder installed. Checking every 15 min, emailing ' + who);
}

/** Runs on the trigger. Emails once per missed slot, never twice. */
function checkShotDue() {
  if (!ALERT_EMAIL) return;
  const now = new Date();
  const rows = readRows();

  // Every insulin time logged, as real dates, so a slot just before midnight
  // can still be judged from the far side of midnight.
  const shots = [];
  rows.forEach(function (r) {
    if (String(r.type).trim().toLowerCase() !== 'insulin') return;
    const d = parseRowDate_(r.date, r.time);
    if (d) shots.push(d);
  });

  slotsToCheck_(now).forEach(function (slot) {
    const from = new Date(slot.at.getTime() - LOOKBACK_H * 3600e3);
    const covered = shots.some(function (d) { return d >= from && d <= now; });
    if (covered) return;

    const key = 'alerted:' + slot.key;
    const props = PropertiesService.getScriptProperties();
    if (props.getProperty(key)) return;

    const last = shots.length ? new Date(Math.max.apply(null, shots)) : null;
    const lateMin = Math.round((now - slot.at) / 60000);
    MailApp.sendEmail(
      ALERT_EMAIL,
      'Zuse: ' + slot.name + ' insulin not logged',
      'The ' + slot.name + ' shot was due at ' + slot.hhmm + '.\n' +
      'It is now ' + fmtLocal_(now) + ' — ' + lateMin + ' minutes past, and nothing is logged.\n\n' +
      (last ? 'Last insulin logged: ' + fmtLocal_(last) + '\n\n' : 'No insulin has ever been logged.\n\n') +
      'If he has had it, log it in the app so this stops chasing you.\n' +
      '(One email per missed shot. Nothing else is sent.)'
    );
    props.setProperty(key, '1');
    cleanupAlertKeys_(props);
  });
}

/** Yesterday's and today's slots, keeping only those now sitting in the
 *  "late enough to mention, not so late it's history" window. */
function slotsToCheck_(now) {
  const out = [];
  [-1, 0].forEach(function (dayOffset) {
    [['morning', AM_TIME], ['night', PM_TIME]].forEach(function (pair) {
      const parts = String(pair[1]).split(':');
      const at = new Date(now.getFullYear(), now.getMonth(), now.getDate() + dayOffset,
                          Number(parts[0]), Number(parts[1]), 0);
      const late = now - at;
      if (late < GRACE_MIN * 60e3) return;
      if (late > ALERT_WINDOW_H * 3600e3) return;
      out.push({
        name: pair[0],
        hhmm: pair[1],
        at: at,
        key: Utilities.formatDate(at, Session.getScriptTimeZone(), 'yyyy-MM-dd') + ':' + pair[0]
      });
    });
  });
  return out;
}

/** 'dd/MM/yyyy' + 'h:mm am' -> Date, or null if either is unparseable. */
function parseRowDate_(dmy, t12) {
  const dm = String(dmy).trim().match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/);
  const tm = String(t12).trim().match(/^(\d{1,2}):(\d{2})\s*(am|pm)$/i);
  if (!dm || !tm) return null;
  let h = Number(tm[1]) % 12;
  if (/pm/i.test(tm[3])) h += 12;
  return new Date(Number(dm[3]), Number(dm[2]) - 1, Number(dm[1]), h, Number(tm[2]), 0);
}

function fmtLocal_(d) {
  return Utilities.formatDate(d, Session.getScriptTimeZone(), 'EEE d MMM, h:mm a');
}

/** Script properties are capped, and one key per slot adds up. Keep a fortnight. */
function cleanupAlertKeys_(props) {
  const cutoff = Utilities.formatDate(new Date(Date.now() - 14 * 864e5),
                                      Session.getScriptTimeZone(), 'yyyy-MM-dd');
  const all = props.getProperties();
  Object.keys(all).forEach(function (k) {
    if (k.indexOf('alerted:') === 0 && k.substring(8, 18) < cutoff) props.deleteProperty(k);
  });
}

/* ==========================================================================
 * MONTHLY REPORT (written by Gemini, emailed to you)
 *
 * On the 1st of each month this reads the last 30 days off the sheet, plus the
 * 30 days before that for comparison, and emails you a few paragraphs of plain
 * English: what moved, what didn't, what's worth mentioning to the vet.
 *
 * Only aggregates are sent to the model — totals, averages, weigh-ins, glucose
 * readings and the incident list. Never the whole sheet. It is cheaper, and
 * there is far less for the model to get wrong.
 *
 * The API key is NOT in this file and must never be. It lives in the script's
 * own properties, behind your Google account.
 *
 * SETUP (one-time):
 * 1. Get a free key at https://aistudio.google.com/apikey
 * 2. Apps Script editor -> Project Settings (gear) -> Script Properties ->
 *    Add script property.  Property: GEMINI_API_KEY   Value: <your key>
 * 3. Make sure ALERT_EMAIL above is set — the report goes to the same address.
 * 4. Run "runMonthlyReportNow" once to check it works. You should get an email
 *    within a minute or so.
 * 5. Run "installMonthlyReport" to put it on the monthly schedule.
 *
 * If the key is missing or Google's API is down, you still get the email — it
 * just carries the raw numbers and says what went wrong. A report that fails
 * quietly is worse than no report.
 * ========================================================================== */

const GEMINI_MODEL = 'gemini-3.6-flash';
const REPORT_DAYS = 30;      // length of each comparison window
const REPORT_HOUR = 9;       // hour of the 1st to send on (24h, script timezone)

/** Puts the report on the 1st of every month. Run this by hand once. */
function installMonthlyReport() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'sendMonthlyReport') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('sendMonthlyReport')
    .timeBased().onMonthDay(1).atHour(REPORT_HOUR).create();
  Logger.log('Monthly report installed: 1st of each month, around ' +
             REPORT_HOUR + ':00, to ' + (ALERT_EMAIL || '(nobody — ALERT_EMAIL is blank)'));
}

/** Same thing, on demand. Use this to test the setup. */
function runMonthlyReportNow() {
  sendMonthlyReport();
  Logger.log('Report sent to ' + ALERT_EMAIL);
}

function sendMonthlyReport() {
  if (!ALERT_EMAIL) {
    Logger.log('ALERT_EMAIL is blank — nothing sent.');
    return;
  }
  const stats = buildReportStats_(REPORT_DAYS);
  let body, note = '';
  try {
    const prose = askGemini_(reportPrompt_(stats));
    if (!prose) {
      note = 'No GEMINI_API_KEY is set in Script Properties, so this month has ' +
             'the numbers but no write-up.\n\n';
      body = note + stats.text;
    } else {
      body = prose + '\n\n' +
             '-------------------------------------------\n' +
             'The figures this was written from:\n\n' + stats.text;
    }
  } catch (err) {
    // Never swallow it — send the numbers and say what broke.
    body = 'The write-up failed this month (' + err + '), so here are the raw ' +
           'figures instead.\n\n' + stats.text;
  }
  MailApp.sendEmail(ALERT_EMAIL, 'Zuse: monthly report — ' + stats.label, body);
}

/* ---------------- the numbers ---------------- */

/** Two windows: the last `days`, and the `days` before that. */
function buildReportStats_(days) {
  const rows = readRows();
  const now = new Date();
  const startA = new Date(now.getTime() - days * 864e5);
  const startB = new Date(now.getTime() - 2 * days * 864e5);
  const A = bucket_(rows, startA, now);
  const B = bucket_(rows, startB, startA);

  const L = [];
  L.push('THIS PERIOD  (A): ' + fmtDay_(startA) + ' to ' + fmtDay_(now) + ', ' + days + ' days');
  L.push('PREVIOUS     (B): ' + fmtDay_(startB) + ' to ' + fmtDay_(startA) + ', ' + days + ' days');
  L.push('');
  L.push('TOTALS BY TYPE  (A now, B before)');

  const names = {};
  [A, B].forEach(function (b) {
    Object.keys(b.totals).forEach(function (k) { names[k] = b.totals[k].name; });
  });
  Object.keys(names).sort().forEach(function (k) {
    const a = A.totals[k] || { qty: 0, count: 0 };
    const b = B.totals[k] || { qty: 0, count: 0 };
    L.push('  ' + names[k] + ': ' +
      round_(a.qty) + ' over ' + a.count + ' entries (' + perDay_(a.qty, days) + '/day)' +
      '   was  ' +
      round_(b.qty) + ' over ' + b.count + ' entries (' + perDay_(b.qty, days) + '/day)');
  });

  L.push('');
  L.push('FREE-TEXT ROWS, no matching type (A): ' + A.freeText.length);
  A.freeText.forEach(function (t) { L.push('  ' + t); });

  L.push('');
  L.push('WEIGHT (A): ' + (A.weights.length ? A.weights.join(', ') : 'no weigh-ins'));
  L.push('WEIGHT (B): ' + (B.weights.length ? B.weights.join(', ') : 'no weigh-ins'));
  L.push('GLUCOSE (A): ' + (A.glucose.length ? A.glucose.join(', ') : 'no readings'));
  L.push('GLUCOSE (B): ' + (B.glucose.length ? B.glucose.join(', ') : 'no readings'));

  L.push('');
  L.push('INCIDENTS — anything reading as sick, diarrhea, vomiting, runny/soft poop or a pee accident');
  L.push('  A: ' + A.incidents.length + (A.incidents.length ? '' : ' (none)'));
  A.incidents.forEach(function (s) { L.push('    ' + s); });
  L.push('  B: ' + B.incidents.length + (B.incidents.length ? '' : ' (none)'));
  B.incidents.forEach(function (s) { L.push('    ' + s); });

  if (A.urine.length) {
    L.push('');
    L.push('URINE TESTS (A):');
    A.urine.forEach(function (s) { L.push('  ' + s); });
  }

  return { text: L.join('\n'), label: fmtDay_(startA) + ' to ' + fmtDay_(now) };
}

/** Everything in [from, to), keyed by lowercased sheet type. */
/* The sheet's TYPE column has been used as free text for years, so the same
 * thing appears as "Chicken Slice" and "Chicken Slice \uD83C\uDF57", and real events hide
 * inside one-off phrases like "diarrhea still". Everything below normalises
 * before counting; keep this table in step with TYPE_ALIASES in the app. */
const KNOWN_TYPES = {
  'canned food': 'Canned Food', 'dry food': 'Dry Food',
  'insulin': 'Insulin', 'samylin': 'Samylin',
  'prednisolone': 'Prednisolone', 'prednidale': 'Prednisolone',
  'paracetamol': 'Paracetamol', 'synulox 250mg': 'Synulox 250mg',
  'cerenia 24mg': 'Cerenia 24mg', 'promax': 'Promax',
  'chicken slice': 'Chicken Slice', 'carrot': 'Carrot',
  'treat': 'Treat', 'treats': 'Treat',
  'walk': 'Walk', 'bath': 'Bath', 'note': 'Note',
  'weight': 'Weight', 'glucose': 'Glucose', 'urine test': 'Urine test',
  'vets visit': 'Vets Visit',
  'wormer / flea': 'Wormer / Flea', 'frontline wormer': 'Wormer / Flea',
  'poop': 'Poop', 'wee wee': 'Wee Wee', 'weewee': 'Wee Wee',
  'pee accident': 'Pee accident', 'was sick': 'Was sick', 'diarrhea': 'Diarrhea'
};
/* Reported on their own further down, so they must not also appear as a total. */
const NOT_A_TOTAL = ['Weight', 'Glucose', 'Urine test', 'Vets Visit', 'Note'];
/* Substring match. The real rows read "diarrhea still", "wee we/runny poop" and
 * so on; exact matching undercounted these by more than half. */
const INCIDENT_RE = /sick|diarrh|runny poop|soft poop|pee accident|vomit/i;

/** Strip emoji and stray punctuation, collapse spaces, then look it up. */
function canonType_(raw) {
  const clean = String(raw).replace(/[^\x20-\x7E]/g, '').replace(/\s+/g, ' ').trim().toLowerCase();
  return KNOWN_TYPES[clean] || null;
}

/** Identity of an incident, independent of which row it arrived on: the day,
 *  plus the words, with the type dropped when it is only the generic "Note". */
function incidentKey_(day, raw, notes, name) {
  const body = ((name === 'Note' ? '' : raw) + ' ' + notes)
    .replace(/[^a-zA-Z0-9 ]/g, ' ')
    .replace(/\s+/g, ' ').trim().toLowerCase();
  return day + '|' + body;
}

function bucket_(rows, from, to) {
  const out = { totals: {}, weights: [], glucose: [], incidents: [], urine: [], freeText: [] };
  const seenWeight = {};
  const incidentSeen = {};
  rows.forEach(function (r) {
    const when = parseRowDate_(r.date, r.time);
    if (!when || when < from || when >= to) return;
    const raw = String(r.type).trim();
    const name = canonType_(raw);
    const n = numFrom_(r.qty);
    const day = fmtDay_(when);

    // An unrecognised TYPE is a hand-typed note, not a category. Treating it as
    // one produced forty rows of noise and split the totals that mattered.
    if (!name) {
      out.freeText.push(day + ' — ' + raw + (r.notes ? ': ' + r.notes : ''));
    } else if (NOT_A_TOTAL.indexOf(name) === -1) {
      const t = out.totals[name] || (out.totals[name] = { name: name, qty: 0, count: 0 });
      t.qty += n;
      t.count++;
    }

    if (name === 'Weight') {
      const k = day + '|' + n;                    // the sheet carries duplicate weigh-ins
      if (!seenWeight[k]) { seenWeight[k] = 1; out.weights.push(n + ' kg on ' + day); }
    } else if (name === 'Glucose') {
      out.glucose.push(n + ' on ' + day);
    } else if (name === 'Urine test') {
      out.urine.push(day + ' — ' + (r.notes || 'no values recorded'));
    }

    // Tested against the raw text AND the notes, so hand-typed events count too.
    const notes = r.notes ? String(r.notes) : '';
    if (INCIDENT_RE.test(raw) || INCIDENT_RE.test(notes)) {
      // Name the event, not the row it arrived on. A Canned Food row whose note
      // says "soft poop" is a soft poop, not a feeding.
      const label = INCIDENT_RE.test(raw)
        ? raw + (notes ? ': ' + notes : '')
        : (notes || raw);
      // Unrecognised sheet types get pulled into the app as notes and pushed
      // back as type "Note", so the same event is in the sheet twice. Collapse
      // them, keeping whichever copy carries the more specific type.
      const key = incidentKey_(day, raw, notes, name);
      const better = (name !== 'Note' && name !== null) || !incidentSeen[key];
      if (!incidentSeen[key] || better) incidentSeen[key] = day + ' — ' + label;
    }
  });
  Object.keys(incidentSeen).forEach(function (k) { out.incidents.push(incidentSeen[k]); });
  // Both lists lead with the date, so a plain sort puts them in date order.
  out.incidents.sort();
  out.freeText.sort();
  return out;
}

/** Sheet quantities mix plain numbers with number+unit text ("8 Units"). */
function numFrom_(v) {
  if (v === '' || v == null) return 0;
  const m = String(v).trim().match(/^(-?\d+(?:\.\d+)?)/);
  return m ? Number(m[1]) : 0;
}
function round_(n) { return Math.round(n * 100) / 100; }
function perDay_(n, days) { return (n / days).toFixed(2); }
function fmtDay_(d) {
  return Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd');
}

/* ---------------- the model ---------------- */

function reportPrompt_(stats) {
  return [
    'You are writing a short monthly update for the owner of Zuse, a diabetic dog,',
    'from his care log. The owner will read it, and may forward it to their vet.',
    '',
    'Write 3 to 5 short paragraphs of plain English. Rules:',
    '- Quote the actual numbers. "insulin averaged 8.4 units a day, up from 7.9" —',
    '  never "insulin usage increased".',
    '- Compare period A against period B and say which way things moved.',
    '- If something did not change, say so plainly. Do not manufacture a trend',
    '  out of noise, and do not pad the report to fill space.',
    '- Call out anything a vet would want flagged: weight moving, glucose readings,',
    '  a cluster of sick or diarrhea entries, walks or food dropping off.',
    '- Missing data is a fact worth stating (e.g. no weigh-in this period), not a',
    '  gap to guess across.',
    '- The free-text rows were typed by hand rather than picked from a list. Read',
    '  them for anything the totals miss, but do not count them twice: the ones',
    '  that matter already appear in the incident list.',
    '',
    'Hard limits:',
    '- Do NOT suggest, adjust, or comment on insulin doses or any medication dose.',
    '- Do NOT diagnose, and do NOT give clinical advice or reassurance.',
    '- You are describing what the log says. Anything beyond that is the vet\'s job.',
    '  If the numbers look concerning, the only recommendation you may make is to',
    '  raise it with the vet.',
    '',
    'No headings, no bullet lists, no preamble like "Here is your report".',
    'Start with the sentence that matters most.',
    '',
    'THE LOG:',
    '',
    stats.text
  ].join('\n');
}

/* A busy model answers 503, and this job only runs once a month — one attempt
 * is not enough. Back off and try again; 5 + 10 + 20 seconds is nothing against
 * the six-minute script limit. */
const RETRY_CODES = [429, 500, 502, 503, 504];
const RETRIES = 4;

function askGemini_(prompt) {
  const key = PropertiesService.getScriptProperties().getProperty('GEMINI_API_KEY');
  if (!key) return null;   // caller turns this into the numbers-only email
  const url = 'https://generativelanguage.googleapis.com/v1beta/models/' +
              GEMINI_MODEL + ':generateContent?key=' + encodeURIComponent(key);
  const params = {
    method: 'post',
    contentType: 'application/json',
    payload: JSON.stringify({
      contents: [{ parts: [{ text: prompt }] }],
      // Deliberately no output cap: a busy month is a long write-up, and a
      // limit here truncates it mid-sentence rather than failing loudly.
      generationConfig: { temperature: 0.3 }
    }),
    muteHttpExceptions: true
  };

  let last = '';
  for (let attempt = 0; attempt < RETRIES; attempt++) {
    const res = UrlFetchApp.fetch(url, params);
    const code = res.getResponseCode();
    const text = res.getContentText();

    if (code === 200) {
      const data = JSON.parse(text);
      const cand = data && data.candidates && data.candidates[0];
      const parts = cand && cand.content && cand.content.parts;
      // With no cap, an empty answer means the model spent its whole budget
      // thinking. Name the reason so the failure email says why.
      if (!parts || !parts.length) {
        throw new Error('Gemini returned no text (finishReason ' +
          ((cand && cand.finishReason) || 'none') + ') — ' + text.slice(0, 300));
      }
      const out = parts.map(function (p) { return p.text || ''; }).join('').trim();
      if (!out) throw new Error('Gemini returned an empty message');
      return out;
    }

    last = 'Gemini HTTP ' + code + ' — ' + text.slice(0, 300);
    // A bad key or a retired model will never fix itself; don't burn four tries.
    if (RETRY_CODES.indexOf(code) === -1) throw new Error(last);
    if (attempt < RETRIES - 1) Utilities.sleep(Math.pow(2, attempt) * 5000);
  }
  throw new Error(last + '  (gave up after ' + RETRIES + ' attempts)');
}


/* ==========================================================================
 * STOCK - running out, before you run out
 *
 * The app counts down from what you say you have in hand. This mirrors those
 * numbers onto a "Stock" tab, then once a day works out how fast each one is
 * actually being used (from the sheet's own rows) and tells you when it is
 * about to run out.
 *
 * You get told twice, and only once per restock:
 *   - an email now, when it drops under a week of supply. Insulin is the one
 *     exception: it is ordered by the bottle, so it warns at one bottle (400
 *     units) left, which is roughly seven weeks rather than seven days.
 *   - a Google Calendar event on the day it is predicted to run out, with a
 *     popup reminder. Because the script runs as you, that lands on any phone
 *     signed into the same Google account.
 *
 * SETUP (one-time):
 * 1. ALERT_EMAIL above must be set - the same one the shot reminder uses.
 * 2. Sync once from the app. That creates the "Stock" tab.
 * 3. In the editor pick "installStockAlerts" and Run. Authorize when asked -
 *    this is the run that asks for calendar access, which is why it is done
 *    by hand and never from the web app.
 *
 * To turn it off: Triggers (clock icon) -> delete checkStock.
 * ========================================================================== */

const STOCK_SHEET = 'Stock';
const STOCK_LOW_DAYS = 7;    // warn once supply drops under this many days
const STOCK_WINDOW = 14;     // days of history the daily burn rate averages over
const STOCK_HOUR = 8;        // hour of the day the check runs (24h, script timezone)

/* Insulin is ordered by the bottle and a week's notice is not enough to get one,
 * so it warns on what is left rather than on days. 400 units is one bottle.
 * Keyed by the ITEM name on the Stock tab; keep in step with LOW_LEFT in the app. */
const STOCK_LOW_LEFT = { 'Insulin': 400 };

/* Items used one-per-entry of another type rather than by quantity. A syringe is
 * never logged, but one goes with every insulin shot, so its burn rate is the
 * number of Insulin rows and not the sum of their units. Matches PER_SHOT. */
const STOCK_PER_ROW = { 'Syringes': 'Insulin' };

/* A second, separate alert: Prednisolone and Syringes also go straight to the
 * vet asking them to reorder, once per restock cycle, independent of the
 * email above. Keyed by the ITEM name on the Stock tab; keep VET_REORDER in
 * step with the app's own VET_REORDER list. */
const VET_EMAIL = 'hellopoole@natterjacksvet.com';
const VET_PHONE = '07496 751 666';
const VET_REORDER_DAYS = 10;
const VET_REORDER = { 'Prednisolone': { qty: 30, unit: 'tablets' }, 'Syringes': { qty: 60, unit: 'syringes' } };

/** The two rules in one place, so the email and the app agree on "low". */
function stockIsLow_(it, f) {
  const cap = STOCK_LOW_LEFT[it.name];
  if (cap !== undefined) return f.left <= cap;
  return f.days !== null && f.days <= STOCK_LOW_DAYS;
}

/** Installs (or reinstalls) the daily stock check. Run this by hand once. */
function installStockAlerts() {
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === 'checkStock') ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger('checkStock').timeBased().atHour(STOCK_HOUR).everyDays(1).create();
  Logger.log('Stock check installed, running daily at ' + STOCK_HOUR + ':00.');
}

/** Same thing on demand, for checking the setup works. */
function checkStockNow() {
  const n = checkStock();
  Logger.log(n ? 'Alerted on ' + n + ' item(s).' : 'Nothing is running low.');
}

/** Runs on the trigger. Emails once per item per restock, never twice. */
function checkStock() {
  if (!ALERT_EMAIL) return 0;
  const items = readStockTab_();
  if (!items.length) return 0;          // no Stock tab yet, or nothing tracked

  const rows = readRows();
  const now = new Date();
  const props = PropertiesService.getScriptProperties();
  const low = [];
  const reorders = [];

  items.forEach(function (it) {
    const f = stockForecast_(it, rows, now);
    if (stockIsLow_(it, f)) {
      // One alert per restock cycle: entering a new amount moves `since`, which
      // makes a new key, which lets it speak up again.
      const key = 'stock:' + it.name + '|' + it.since;
      if (!props.getProperty(key)) {
        props.setProperty(key, '1');
        low.push(Object.assign({}, it, f));
      }
    }
    // The vet-reorder email is separate from the one above: its own item
    // list, its own day threshold, its own dedupe key, so cancelling or
    // suppressing one never touches the other. Sent after emailLowStock_
    // below, not here, so the order of outgoing mail stays predictable.
    const reorder = VET_REORDER[it.name];
    if (reorder && f.days !== null && f.days <= VET_REORDER_DAYS) {
      const vkey = 'vetorder:' + it.name + '|' + it.since;
      if (!props.getProperty(vkey)) {
        props.setProperty(vkey, '1');
        reorders.push({ name: it.name, reorder: reorder });
      }
    }
  });

  cleanupStockKeys_(props, items);

  if (low.length) {
    emailLowStock_(low);
    low.forEach(function (x) { addReorderReminder_(x); });
  }
  reorders.forEach(function (x) { emailVetReorder_(x.name, x.reorder); });
  return low.length + reorders.length;
}

/** The Stock tab, or [] when it does not exist yet. Never throws: a missing
 *  tab on a fresh install must not make this fail every morning. */
function readStockTab_() {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(STOCK_SHEET);
  if (!sh) return [];
  const v = sh.getDataRange().getValues();
  const out = [];
  for (let i = 1; i < v.length; i++) {          // row 1 is the header
    const name = String(v[i][0]).trim();
    if (!name) continue;
    out.push({
      name: name,
      qty: numFrom_(v[i][1]),
      unit: String(v[i][2] === undefined ? '' : v[i][2]).trim(),
      since: formatDate_(v[i][3])
    });
  }
  return out;
}

/** What is left, how fast it is going, and when it runs out.
 *  Same model as the app: remaining = what was entered, minus everything
 *  logged on or after that day. Edits and deletions correct themselves. */
function stockForecast_(it, rows, now) {
  const since = parseRowDate_(it.since, '12:00 am');
  const from = new Date(now.getTime() - (STOCK_WINDOW - 1) * 864e5);
  from.setHours(0, 0, 0, 0);

  // A syringe has no rows of its own; it is spent one per insulin shot, so the
  // row it is counted from is Insulin's and each one counts as exactly 1.
  const src = STOCK_PER_ROW[it.name] || it.name;
  const perRow = !!STOCK_PER_ROW[it.name];

  let used = 0, recent = 0;
  rows.forEach(function (r) {
    if (canonType_(r.type) !== src) return;
    const d = parseRowDate_(r.date, r.time);
    if (!d) return;
    const n = perRow ? 1 : numFrom_(r.qty);
    if (since && d >= since) used += n;
    if (d >= from) recent += n;
  });

  // Fixed divisor, not "days that happen to have an entry": dividing by the
  // latter reads a twice-weekly tablet as a daily one and halves the estimate.
  const rate = recent / STOCK_WINDOW;
  const left = round_(it.qty - used);
  const days = rate > 0 ? Math.floor(Math.max(0, left) / rate) : null;
  return {
    left: left,
    used: round_(used),
    rate: round_(rate),
    days: days,
    outOn: days === null ? null : new Date(now.getTime() + days * 864e5)
  };
}

function emailLowStock_(low) {
  const names = low.map(function (x) { return x.name; }).join(', ');
  const lines = low.map(function (x) {
    const u = x.unit ? ' ' + x.unit : '';
    // The "one bottle left" rule can fire with nothing logged for a fortnight,
    // and there is no forecast to print then.
    return '- ' + x.name + ': ' + x.left + u + ' left, about ' + x.rate + u +
           ' a day - ' + (x.days === null ? 'nothing logged lately, so no estimate' :
             x.days === 0 ? 'gone now' :
             x.days + ' day' + (x.days === 1 ? '' : 's') +
             (x.outOn ? ', around ' + fmtDay_(x.outOn) : ''));
  });
  MailApp.sendEmail(ALERT_EMAIL, 'Zuse: running low on ' + names,
    'Time to reorder.\n\n' + lines.join('\n') + '\n\n' +
    'Worked out from the last ' + STOCK_WINDOW + ' days of entries on the sheet, ' +
    'counting down from the amount last entered in the app.\n' +
    'A calendar reminder has been added for each one with a predicted day.\n');
}

/** Short, polite ask to the vet to reorder and call when it's ready. Separate
 *  from emailLowStock_ above: different recipient, different day threshold,
 *  different wording. */
function emailVetReorder_(name, reorder) {
  MailApp.sendEmail(VET_EMAIL, 'Reorder request: ' + name,
    'Hi,\n\nCould you please order ' + reorder.qty + ' ' + reorder.unit + ' of ' + name + ' for Zuse?\n\n' +
    'Please call me on ' + VET_PHONE + ' when it is ready to collect.\n\nThanks!\n');
}

/** A calendar event on your own Google account, so the reminder reaches the
 *  phone signed into it. Only ever called from the daily trigger - see the
 *  note in doPost about why calendar access must stay off the sync path. */
function addReorderReminder_(x) {
  if (x.days === null) return;   // no burn rate, so no day to put it on
  try {
    const cal = CalendarApp.getDefaultCalendar();
    if (!cal) return;
    const when = new Date();
    when.setDate(when.getDate() + Math.max(1, x.days));   // tomorrow at the earliest
    when.setHours(9, 0, 0, 0);
    const title = 'Zuse: ' + x.name + ' runs out';
    // Don't stack copies if the trigger ever fires twice for one cycle.
    const clash = cal.getEventsForDay(when, { search: title });
    if (clash && clash.length) return;
    const u = x.unit ? ' ' + x.unit : '';
    const ev = cal.createEvent(title, when, new Date(when.getTime() + 15 * 6e4), {
      description: x.left + u + ' left as of ' + fmtDay_(new Date()) + ', going at about ' +
                   x.rate + u + ' a day over the last ' + STOCK_WINDOW + ' days.'
    });
    ev.addPopupReminder(0);
  } catch (err) {
    // A missing calendar scope must not lose the email that already went out.
    Logger.log('Calendar reminder skipped: ' + err);
  }
}

/** Drops the "already told you" keys for restocks that are no longer current,
 *  so entering a new amount lets the alert fire again. */
function cleanupStockKeys_(props, items) {
  const live = {};
  items.forEach(function (it) {
    live['stock:' + it.name + '|' + it.since] = 1;
    live['vetorder:' + it.name + '|' + it.since] = 1;
  });
  Object.keys(props.getProperties()).forEach(function (k) {
    if ((k.indexOf('stock:') === 0 || k.indexOf('vetorder:') === 0) && !live[k]) props.deleteProperty(k);
  });
}

/** Mirrors the app's stock levels onto the Stock tab. Called from doPost. */
function writeStockTab_(items) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(STOCK_SHEET) || ss.insertSheet(STOCK_SHEET);
  sh.clear();
  const head = ['ITEM', 'IN HAND', 'UNIT', 'COUNTING FROM', 'UPDATED'];
  const stamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd/MM/yyyy HH:mm');
  const body = items.map(function (it) {
    return [String(it.name || ''), Number(it.qty) || 0, String(it.unit || ''),
            String(it.since || ''), stamp];
  });
  sh.getRange(1, 1, 1, head.length).setValues([head]);
  // COUNTING FROM is dd/MM/yyyy. This tab is created from scratch, so unlike
  // the main sheet's date column it has no format history - left alone, Sheets
  // would read "05/08/2026" through the spreadsheet's locale and could land on
  // 8 May. Pinning the column to text keeps it the string that was written.
  sh.getRange(2, 4, Math.max(body.length, 1), 1).setNumberFormat('@');
  if (body.length) sh.getRange(2, 1, body.length, head.length).setValues(body);
  return body.length;
}
