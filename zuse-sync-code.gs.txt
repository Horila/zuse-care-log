const SECRET = 'CHANGE_ME_TO_YOUR_OWN_SECRET';
const SHEET_NAME = 'Zuse Schedule';

function doGet(e) {
  if (!e.parameter.secret || e.parameter.secret !== SECRET) {
    return jsonOut({ error: 'unauthorized' });
  }
  const action = e.parameter.action || 'list';
  if (action === 'list') {
    return jsonOut({ rows: readRows() });
  }
  return jsonOut({ error: 'unknown action' });
}

function doPost(e) {
  let body;
  try {
    body = JSON.parse(e.postData.contents);
  } catch (err) {
    return jsonOut({ error: 'bad json' });
  }
  if (!body.secret || body.secret !== SECRET) {
    return jsonOut({ error: 'unauthorized' });
  }
  if (body.action === 'append') {
    const added = appendRows(body.rows || []);
    return jsonOut({ added: added });
  }
  return jsonOut({ error: 'unknown action' });
}

function getSheet_() {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  if (!sh) throw new Error('Sheet "' + SHEET_NAME + '" not found');
  return sh;
}

/** Reads every data row (skipping the 2 header rows), carrying the date
 *  forward for rows where the DATE cell is blank (same-day continuation). */
function readRows() {
  const sh = getSheet_();
  const values = sh.getDataRange().getValues();
  const out = [];
  let curDate = '';
  for (let i = 2; i < values.length; i++) {
    const row = values[i];
    const dateCell = row[1];
    const timeCell = row[2];
    const type = row[3];
    const qty = row[4];
    const notes = row[5];
    if (dateCell) curDate = formatDate_(dateCell);
    if (!type) continue; // blank/free-text rows with no TYPE are skipped
    out.push({
      date: curDate,
      time: formatTime_(timeCell),
      type: String(type),
      qty: (qty === '' || qty == null) ? '' : qty,
      notes: notes ? String(notes) : ''
    });
  }
  return out;
}

/** Appends rows at the bottom, keeping the "blank date = same day as
 *  previous row" convention used throughout the sheet. */
function appendRows(rows) {
  const sh = getSheet_();
  const lastRow = sh.getLastRow();
  let lastDate = '';
  if (lastRow >= 3) {
    for (let r = lastRow; r >= 3; r--) {
      const d = sh.getRange(r, 2).getValue();
      if (d) { lastDate = formatDate_(d); break; }
    }
  }
  const out = [];
  rows.forEach(function (r) {
    const sameDate = (r.date === lastDate);
    out.push(['', sameDate ? '' : r.date, r.time, r.type, r.qty, r.notes || '']);
    lastDate = r.date;
  });
  if (out.length) {
    sh.getRange(sh.getLastRow() + 1, 1, out.length, 6).setValues(out);
  }
  return out.length;
}

function formatDate_(v) {
  if (Object.prototype.toString.call(v) === '[object Date]') {
    return Utilities.formatDate(v, Session.getScriptTimeZone(), 'dd/MM/yyyy');
  }
  return String(v).trim();
}
function formatTime_(v) {
  if (Object.prototype.toString.call(v) === '[object Date]') {
    return Utilities.formatDate(v, Session.getScriptTimeZone(), 'h:mm a').toLowerCase();
  }
  return String(v).trim().toLowerCase();
}

function jsonOut(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
